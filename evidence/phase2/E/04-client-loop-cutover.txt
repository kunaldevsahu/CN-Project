# Date:      2026-11-25 15:34:04 IST
# Run on:    Kunals-MacBook-Pro-3 10.7.30.71  (script: scripts/phase2/E-cutover.sh )
# Client:    mac1 = Vaibhavs-MacBook-Air-5 10.7.17.233
# server = answer of DNS server 10.7.17.233 right now (dig over TCP, no cache); curl-> = IP curl actually connected to (OS cache)
# Record change app.team1.test -> 10.7.3.175 at ~t+20s (see 03-events.txt / 05-set-record-to-standby.txt)
# Every command is printed before its output: '$ ' = this Mac, '[mac1]$ ' = the client.

[mac1]$ t0=$(date +%s); end=$((SECONDS+90)); i=0
while [ $SECONDS -lt $end ]; do i=$((i+1))
  srv=$(dig +tcp +short +time=1 +tries=1 @10.7.17.233 $NAME 2>/dev/null | grep -E '^[0-9.]+$' | head -1)
  h=$(curl -s -D - -o /dev/null --max-time 5 -w 'IP=%{remote_ip} CODE=%{http_code}' https://$NAME/api/status)
  ip=$(echo "$h" | sed -n 's/.*IP=\([0-9.]*\).*/\1/p'); code=$(echo "$h" | sed -n 's/.*CODE=\([0-9]*\).*/\1/p')
  flag=""; [ -n "$srv" ] && [ -n "$ip" ] && [ "$srv" != "$ip" ] && flag="  <- OS cache still has the old IP"
  printf '%s  t+%-3s server=%-11s curl->%-11s http=%-3s x-edge=%-13s x-backend=%-2s%s\n' "$(tsm)" $(($(date +%s)-t0)) "${srv:--}" "${ip:--}" "${code:--}" "$(echo "$h" | hv x-edge)" "$(echo "$h" | hv x-backend)" "$flag"
  sleep 1
done
