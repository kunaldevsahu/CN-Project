## Same ETag from both backends (same body), so a 304 works whichever backend answers
[mac1]$ for i in 1 2 3 4; do h=$(curl -sI https://$NAME/api/cached); printf "%s  x-backend=%s  etag=%s\n" "$(date +%T)" "$(echo "$h" | hv x-backend)" "$(echo "$h" | hv etag)"; done

15:05:01  x-backend=A  etag=W/"31-BhpixH3upRuqs+uGPrcnwzO1ql0"
15:05:02  x-backend=B  etag=W/"31-BhpixH3upRuqs+uGPrcnwzO1ql0"

## Conditional request: send the ETag back -> 304 Not Modified, no body
[mac1]$ e=$(curl -sI https://$NAME/api/cached | hv etag); echo "ETag from first response: $e"
for i in 1 2; do echo "$(date +%T)  curl -sI -H \"If-None-Match: $e\" https://$NAME/api/cached"; curl -sI -H "If-None-Match: $e" https://$NAME/api/cached | grep -iE "^HTTP|^etag|^x-backend|^content-length"; done
echo "$(date +%T)  same request with a wrong ETag -> full 200"; curl -sI -H "If-None-Match: \"nope\"" https://$NAME/api/cached | grep -iE "^HTTP"
