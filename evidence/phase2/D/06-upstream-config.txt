upstream team1_backends {
    # Shared memory: the failure counters are shared by all worker processes.
    # 128k, not 64k: nginx needs >= 8 memory pages and Apple silicon pages are 16 KB
    # (64k fails nginx -t with: zone "team1_backends" is too small).
    zone team1_backends 128k;

    # Round robin + PASSIVE health checking (open-source nginx has no active probes):
    # 2 failed attempts (max_fails) within 5 s mark a server down for 5 s (fail_timeout);
    # after that nginx tries it again with the next real request.
    server 10.7.3.175:3001 max_fails=2 fail_timeout=5s;   # Backend A (Mac 3, Himanshu)
    server 127.0.0.1:3002  max_fails=2 fail_timeout=5s;   # Backend B (this Mac, loopback)
}
